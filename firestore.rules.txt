rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /invites/{h} {
      allow get: if request.auth != null;   // 해시를 알아야 조회 가능
      allow list: if false;                 // 열거 금지
      allow write: if false;                // 서버(admin SDK)에서만
    }

    match /students/{uid} {
      allow read, create: if request.auth.uid == uid;
      allow update: if request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['profile','stats','packs','lastUpdated'])
        && request.resource.data.stats.points >= 0
        && request.resource.data.stats.points <= resource.data.stats.points + 2000
        && request.resource.data.stats.points >= resource.data.stats.points - 20000;

      match /cards/{wordId}       { allow read, write: if request.auth.uid == uid; }
      match /customWords/{wordId} { allow read, write: if request.auth.uid == uid; }
      match /sessions/{sid} {
        allow read, create: if request.auth.uid == uid;
        allow update, delete: if false;
      }
    }

    match /feedback/{id} {
      allow create: if true;
      allow read, update, delete: if false;
    }

    match /{document=**} { allow read, write: if false; }
  }
}
