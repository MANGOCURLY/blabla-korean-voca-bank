rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    /* ---------- 초대장 ----------
       문서 ID = sha256(소문자 이메일).

       예전 규칙은 `allow get: if request.auth != null` 이었다. 명단에 있는 학생만
       로그인을 유지할 수 있던 동안에는 그럭저럭 버텼지만, 오픈 모드로 아무나 로그인할 수
       있게 된 지금은 다르다. 남의 이메일을 짐작해 sha256 만 계산하면 그 학생의
       이름·언어·개인 구글시트 주소(vocabCsvUrl)를 그대로 꺼내 갈 수 있다.
       시트는 "웹에 게시" 상태라 주소만 알면 내용이 열린다.

       이제 요청자의 이메일 해시가 문서 ID와 같을 때만 읽힌다 = 자기 것만 읽는다.
       (없는 문서를 읽으면 permission-denied 가 아니라 "문서 없음" 이 돌아오므로
        오픈 모드 분기는 그대로 동작한다.) */
    match /invites/{h} {
      allow get: if request.auth != null
                 && request.auth.token.email != null
                 && hashing.sha256(request.auth.token.email.lower().toUtf8())
                      .toHexString().lower() == h;
      allow list:  if false;   // 열거 금지
      allow write: if false;   // 서버(admin SDK)에서만
    }

    /* ---------- 학생 문서 ---------- */
    match /students/{uid} {
      allow read, create: if request.auth.uid == uid;
      allow update: if request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['profile','stats','packs','lastUpdated'])
        && request.resource.data.stats.points >= 0
        && request.resource.data.stats.points <= resource.data.stats.points + 2000
        && request.resource.data.stats.points >= resource.data.stats.points - 20000;

      /* 카드·개인단어는 자기 것만 읽고 쓴다. 여기까지는 예전과 같다.
         달라진 점: 오픈 모드로 누구나 계정을 만들 수 있게 됐으므로 스키마와 길이를
         못박는다. 안 그러면 남의 Firestore 를 공짜 파일 저장소로 쓸 수 있다
         (문서 하나에 1 MiB 까지 들어간다).
         delete 요청에는 request.resource 가 없으므로 write 에서 분리한다. */
      match /cards/{wordId} {
        allow read, delete: if request.auth.uid == uid;
        allow write: if request.auth.uid == uid
          && request.resource.data.keys().hasOnly(
               ['packId','due','lastReview','interval','ease','state','reps','lapses',
                'gradedCorrect','views','correctStreak','firstSeenAt','masteredAt']);
      }
      match /customWords/{wordId} {
        allow read, delete: if request.auth.uid == uid;
        allow write: if request.auth.uid == uid
          && request.resource.data.keys().hasOnly(['ko','mean','color','order','dateAdded'])
          && request.resource.data.ko    is string && request.resource.data.ko.size()    <= 100
          && request.resource.data.mean  is string && request.resource.data.mean.size()  <= 300
          && request.resource.data.color is string && request.resource.data.color.size() <= 16;
      }
    }

    /* sessions·feedback 컬렉션은 앱이 쓰지 않는다.
       특히 옛 feedback 규칙은 `allow create: if true` 였다 — 로그인조차 없이
       누구나 문서를 만들 수 있는 자리였다. 기능이 실제로 생길 때 다시 연다. */

    match /{document=**} { allow read, write: if false; }
  }
}
